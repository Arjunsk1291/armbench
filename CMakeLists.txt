cmake_minimum_required(VERSION 3.16)
project(armbench)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_planners_ompl REQUIRED)
find_package(pluginlib REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
find_package(gazebo REQUIRED)
add_executable(armbench src/benchmark.cpp src/gazebo_smoke.cpp)
ament_target_dependencies(armbench rclcpp moveit_core moveit_ros_planning moveit_planners_ompl pluginlib urdf srdfdom)
target_include_directories(armbench PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(armbench PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(armbench ${GAZEBO_LIBRARIES})
